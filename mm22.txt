local function SafeCall(fn, ...)

    local args = {...}

    local ok, err = pcall(function() return fn(table.unpack(args)) end)

    if not ok then

        WARN("Callback: " .. tostring(err))

    end

    return ok

end

local function loadSettings()

        if isfile(SETTINGS_FILE) then

            for k, v in pairs(data) do

                if Config[k] ~= nil then Config[k] = v end

    end)

end

local teamList = {}

local teamMap = {}



local function scanTeams()

    local groups = {}

    local order = {}



    for _, player in ipairs(Players:GetPlayers()) do

        local teamName = nil

        

        -- 1) Проверяем атрибут "Team" (основной способ)

        local attr = player:GetAttribute("Team")

        if attr and attr ~= "" then

            teamName = tostring(attr)

        -- 2) Проверяем стандартную команду

        elseif player.Team then

            teamName = player.Team.Name

        -- 3) Проверяем TeamColor

        elseif player.TeamColor then

            local c = player.TeamColor.Color

            local default = Color3.fromRGB(128, 128, 128)

            if c ~= default then

                teamName = player.TeamColor.Name

            end

        end

        

        if teamName then

            if not groups[teamName] then

                groups[teamName] = {name = teamName, count = 0}

                table.insert(order, teamName)

            end

            groups[teamName].count = groups[teamName].count + 1

        end

    end



    table.sort(order, function(a, b) return groups[a].count > groups[b].count end)



    teamList = {}

    teamMap = {}



    for i = 1, math.min(2, #order) do

        local name = order[i]

        teamList[i] = groups[name]

        teamMap[name] = i

    end

    LOG("Команд найдено: " .. #teamList)

end



scanTeams()



local function getTeamIndex(player)

    if not player then return nil end



    local attr = player:GetAttribute("Team")

    if attr and attr ~= "" then

        local idx = teamMap[tostring(attr)]

        if idx then return idx end

    end



    if player.Team then

        local idx = teamMap[player.Team.Name]

        if idx then return idx end

    end



    if player.TeamColor then

        local c = player.TeamColor.Color

        for name, idx in pairs(teamMap) do

            if name == player.TeamColor.Name then

                return idx

            end

        end

    end

    return nil

end



local function getTeamName(player)

    if not player then return "?" end

    local attr = player:GetAttribute("Team")

    if attr and attr ~= "" then return tostring(attr) end

    if player.Team then return player.Team.Name end

    if player.TeamColor then return player.TeamColor.Name end

    return "?"

end

local function PaintToggle(btn, on, label, colorOn)

    if on then

        btn.Text = label .. ": ON"

        btn.BackgroundColor3 = colorOn or Colors.on

        btn.TextColor3 = Color3.new(1, 1, 1)

    else

        btn.Text = label .. ": OFF"

        btn.BackgroundColor3 = Colors.off

        btn.TextColor3 = Color3.new(0.75, 0.75, 0.75)

    end

end



local function MakeToggle(parent, label, order, getter, setter, colorOn)

    local btn = MakeButton(parent, label .. ": OFF", order, function()

        local newState = not getter()

        setter(newState)

        PaintToggle(btn, newState, label, colorOn)

        saveSettings()

    end)

    PaintToggle(btn, getter(), label, colorOn)

    return btn

end

local stickState = {active = false, connection = nil}



local function stopStick()

    if stickState.connection then

        pcall(function() stickState.connection:Disconnect() end)

        stickState.connection = nil

    end

    stickState.active = false

    local char = LP.Character

    if char then

        local hum = char:FindFirstChildOfClass("Humanoid")

        if hum then pcall(function() hum.PlatformStand = false end) end

        for _, part in ipairs(char:GetDescendants()) do

            if part:IsA("BasePart") then

                pcall(function() part.CanCollide = true end)

            end

        end

    end

end



local stickBtn = MakeButton(MurderPage, "Прилипание (2с): OFF", 5, function()

    if stickState.active then

        stopStick()

        PaintToggle(stickBtn, false, "Прилипание (2с)", Colors.danger)

        return

    end

    local target = Config.stickTarget

    if not target or not target.Parent then

        Notify("Ошибка", "Выбери игрока")

        return

    end

    local myChar = LP.Character

    if not myChar then return end

    local myHRP = myChar:FindFirstChild("HumanoidRootPart")

    if not myHRP then return end

    local myHum = myChar:FindFirstChildOfClass("Humanoid")



    for _, part in ipairs(myChar:GetDescendants()) do

        if part:IsA("BasePart") and part.CanCollide then

            part.CanCollide = false

        end

    end

    if myHum then pcall(function() myHum.PlatformStand = true end) end



    local startTime = tick()

    stickState.active = true

    stickState.connection = RunService.RenderStepped:Connect(function()

        pcall(function()

            local tChar = target.Character

            if not tChar then stopStick() return end

            local tHRP = tChar:FindFirstChild("HumanoidRootPart")

            if not tHRP then stopStick() return end

            myHRP.CFrame = tHRP.CFrame * CFrame.new(0, 0, 2)

            myHRP.AssemblyLinearVelocity = Vector3.new(0, 0, 0)

            if tick() - startTime >= Config.stickDuration then

                stopStick()

                PaintToggle(stickBtn, false, "Прилипание (2с)", Colors.danger)

            end

        end)

                end)
        end)
    end)
    PaintToggle(stickBtn, true, "Прилипание (2с)", Colors.danger)
end, Colors.danger)

local espCache = {}
local espDirty = true

local function clearHighlight(player)
    local h = espCache[player]
    if h then
        pcall(function() h:Destroy() end)
        espCache[player] = nil
    end
end

local function applyHighlight(player)
    if player == LP then return end
    local char = player.Character
    if not char then
        clearHighlight(player)
        return
    end

    local idx = getTeamIndex(player)
    local shouldHighlight = false
    if Config.espEnabled then
        if Config.espTeam == 3 then
            shouldHighlight = true
        elseif idx == Config.espTeam then
            shouldHighlight = true
        end
    end

    if not shouldHighlight then
        clearHighlight(player)
        return
    end

    local h = espCache[player]
    if not h or not h.Parent then
        h = Instance.new("Highlight")
        h.Name = "MVS_ESP"
        h.DepthMode = Enum.HighlightDepthMode.Occluded
        h.FillTransparency = 0.55
        h.OutlineTransparency = 0.1
        h.Parent = char
        espCache[player] = h
    end
    h.Adornee = char
    h.FillColor = Config.espColor
    h.OutlineColor = Config.espColor
end

local function hookPlayer(player)
    pcall(function()
        player.CharacterAdded:Connect(function()
            task.wait(0.4)
            espDirty = true
        end)
        player.CharacterRemoving:Connect(function()
            clearHighlight(player)
        end)
        player:GetAttributeChangedSignal("Team"):Connect(function()
            task.wait(0.1)
            espDirty = true
        end)
        player:GetPropertyChangedSignal("Team"):Connect(function()
            task.wait(0.1)
            espDirty = true
        end)
        player:GetPropertyChangedSignal("TeamColor"):Connect(function()
            task.wait(0.1)
            espDirty = true
        end)
    end)
end

for _, p in ipairs(Players:GetPlayers()) do hookPlayer(p) end

local function isVisible(targetChar, fromPart)
    if not Config.wallCheck then return true end
    if not targetChar or not fromPart then return false end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local exclude = {}
    if LP.Character then exclude[1] = LP.Character end
    params.FilterDescendantsInstances = exclude

    for _, part in ipairs(targetChar:GetChildren()) do
        if part:IsA("BasePart") then
            local ray = workspace:Raycast(fromPart.Position, part.Position - fromPart.Position, params)
            if ray and ray.Instance and ray.Instance:IsDescendantOf(targetChar) then
                return true
            end
        end
    end
    return false
end

local function findAimTarget()
    local myChar = LP.Character
    if not myChar then return nil end
    local myHead = myChar:FindFirstChild("Head") or myChar:FindFirstChild("HumanoidRootPart")
    if not myHead then return nil end

    local bestTarget = nil
    local bestDist = math.huge
    local screenCenter = Camera.ViewportSize / 2

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            if getTeamIndex(player) == Config.aimTeam then
                local head = player.Character:FindFirstChild("Head")
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if head and hum and hum.Health > 0 then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                        if dist <= Config.aimFOV and dist < bestDist then
                            if isVisible(player.Character, myHead) then
                                bestTarget = player
                                bestDist = dist
                            end
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

local wallbangActive = false

local function findWallbangTarget()
    local lookVector = Camera.CFrame.LookVector
    local cameraPos = Camera.CFrame.Position
    local bestTarget = nil
    local bestDot = 0.1

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            if getTeamIndex(player) == Config.aimTeam then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local toTarget = hrp.Position - cameraPos
                    if toTarget.Magnitude < 900 then
                        local dot = toTarget.Unit:Dot(lookVector)
                        if dot > bestDot then
                            bestTarget = player
                            bestDot = dot
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

local function doWallbang()
    if wallbangActive then return end
    if not Config.wallbangEnabled then return end

    local target = findWallbangTarget()
    if not target or not target.Character then return end

    local tChar = target.Character
    local tHRP = tChar:FindFirstChild("HumanoidRootPart")
    if not tHRP then return end

    local myTool = LP.Character and LP.Character:FindFirstChildOfClass("Tool")
    if not myTool then
        Notify("Wallbang", "Возьми оружие в руки")
        return
    end

    wallbangActive = true
    LOG("Wallbang: цель " .. target.Name)

    local myChar = LP.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then
        wallbangActive = false
        return
    end

    local savedCF = myHRP.CFrame
    local savedCollisions = {}

    for _, part in ipairs(myChar:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide then
            savedCollisions[part] = true
            part.CanCollide = false
        end
    end

    myHRP.CFrame = tHRP.CFrame * CFrame.new(0, 0, -4)
    myHRP.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    myHRP.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

    task.wait()
    pcall(function() myTool:Activate() end)
    task.wait(0.15)

    pcall(function() myHRP.CFrame = savedCF end)

    for part, _ in pairs(savedCollisions) do
        pcall(function() part.CanCollide = true end)
    end

    LOG("Wallbang: завершено")
    wallbangActive = false
end

local hookedTools = {}
local function hookTool(tool)
    if hookedTools[tool] then return end
    hookedTools[tool] = true
    pcall(function()
        tool.Activated:Connect(function()
            if Config.wallbangEnabled and not wallbangActive then
                task.spawn(doWallbang)
            end
        end)
    end)
    pcall(function()
        tool.TouchTap:Connect(function()
            if Config.wallbangEnabled and not wallbangActive then
                task.spawn(doWallbang)
            end
        end)
    end)
end

local function checkCharacter(char)
    char.ChildAdded:Connect(function(c)
        if c:IsA("Tool") then
            task.wait(0.1)
            hookTool(c)
        end
    end)
    for _, c in ipairs(char:GetChildren()) do
        if c:IsA("Tool") then hookTool(c) end
    end
end

if LP.Character then checkCharacter(LP.Character) end
LP.CharacterAdded:Connect(function(char)
    task.wait(0.5)
    checkCharacter(char)
end)

local shiftLockAlign = nil
local shiftLockAttachment = nil

local function updateShiftLock()
    if not Config.shiftLockEnabled then
        if shiftLockAlign then
            pcall(function() shiftLockAlign:Destroy() end)
            shiftLockAlign = nil
        end
        if shiftLockAttachment then
            pcall(function() shiftLockAttachment:Destroy() end)
            shiftLockAttachment = nil
        end
        return
    end

    local char = LP.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.AutoRotate = false end

    if not shiftLockAttachment or not shiftLockAttachment.Parent then
        shiftLockAttachment = Instance.new("Attachment")
        shiftLockAttachment.Name = "MVS_TurnAttachment"
        shiftLockAttachment.Parent = hrp
    end

    if not shiftLockAlign or not shiftLockAlign.Parent then
        shiftLockAlign = Instance.new("AlignOrientation")
        shiftLockAlign.Name = "MVS_ShiftLock"
        shiftLockAlign.Attachment0 = shiftLockAttachment
        shiftLockAlign.Mode = Enum.OrientationAlignmentMode.OneAttachment
        shiftLockAlign.RigidityEnabled = false
        shiftLockAlign.Responsiveness = 25
        shiftLockAlign.MaxTorque = math.huge
        shiftLockAlign.Enabled = true
        shiftLockAlign.Parent = hrp
    end

    local look = Camera.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude > 0.01 then
        shiftLockAlign.CFrame = CFrame.lookAt(Vector3.zero, flat.Unit)
    end
end

local lastEsp = 0
local lastAim = 0
local lastRainbow = 0
local lastShiftLock = 0
local lastTeams = 0
local cachedAimTarget = nil

RunService.Heartbeat:Connect(function()
    local now = tick()

    if espDirty or (now - lastEsp > 0.3) then
        lastEsp = now
        espDirty = false
        for _, player in ipairs(Players:GetPlayers()) do
            pcall(applyHighlight, player)
        end
    end

    if Config.espRainbow and (now - lastRainbow > 0.1) then
        lastRainbow = now
        Config.espColor = Color3.fromHSV((now * 0.15) % 1, 1, 1)
        espDirty = true
    end

    if Config.aimbotEnabled then
        if now - lastAim > 0.05 then
            lastAim = now
            cachedAimTarget = findAimTarget()
        end
        local target = cachedAimTarget
        if target and target.Character then
            local head = target.Character:FindFirstChild("Head")
            if head then
                local cur = Camera.CFrame
                Camera.CFrame = cur:Lerp(CFrame.new(cur.Position, head.Position), Config.aimSmoothing)
            end
            end
    end

    if Config.shiftLockEnabled and (now - lastShiftLock > 0.02) then
        lastShiftLock = now
        pcall(updateShiftLock)
    end

    if now - lastTeams > 5 then
        lastTeams = now
        pcall(scanTeams)
    end
end)
