local function SafeCall(fn, ...)
    local args = {...}
    local ok, err = pcall(function() return fn(table.unpack(args)) end)
    if not ok then
        WARN("Callback: " .. tostring(err))
    end
    return ok
end

local function saveSettings()
    if not hasFileSystem then
        getgenv().MVS_Settings = Config
        return
    end
    pcall(function()
        local data = {
            aimbotEnabled = Config.aimbotEnabled,

            aimSmoothing = Config.aimSmoothing,

        }
        writefile(SETTINGS_FILE, HttpService:JSONEncode(data))
    end)
end

local function loadSettings()
    if not hasFileSystem then
        if getgenv().MVS_Settings then
            for k, v in pairs(getgenv().MVS_Settings) do
                if Config[k] ~= nil then Config[k] = v end
            end
        end
        return
    end
    pcall(function()
        if isfile(SETTINGS_FILE) then
            local raw = readfile(SETTINGS_FILE)
            local data = HttpService:JSONDecode(raw)
            for k, v in pairs(data) do
                if Config[k] ~= nil then Config[k] = v end
            end
            LOG("Настройки загружены")
        end
    end)
end

-- ================= TEAM MANAGER =================
local teamList = {}
local teamMap = {}

local function scanTeams()
    local groups = {}
    local order = {}

    for _, player in ipairs(Players:GetPlayers()) do
        local teamName = nil
        
        -- 1) Проверяем атрибут "Team" (основной способ)
        local attr = player:GetAttribute("Team")
        if attr and attr ~= "" then
            teamName = tostring(attr)
        -- 2) Проверяем стандартную команду
        elseif player.Team then
            teamName = player.Team.Name
        -- 3) Проверяем TeamColor
        elseif player.TeamColor then
            local c = player.TeamColor.Color
            local default = Color3.fromRGB(128, 128, 128)
            if c ~= default then
                teamName = player.TeamColor.Name
            end
        end
        
        if teamName then
            if not groups[teamName] then
                groups[teamName] = {name = teamName, count = 0}
                table.insert(order, teamName)
            end
            groups[teamName].count = groups[teamName].count + 1
        end
    end

    table.sort(order, function(a, b) return groups[a].count > groups[b].count end)

    teamList = {}
    teamMap = {}

    for i = 1, math.min(2, #order) do
        local name = order[i]
        teamList[i] = groups[name]
        teamMap[name] = i
    end
    LOG("Команд найдено: " .. #teamList)
end

scanTeams()

local function getTeamIndex(player)
    if not player then return nil end

    local attr = player:GetAttribute("Team")
    if attr and attr ~= "" then
        local idx = teamMap[tostring(attr)]
        if idx then return idx end
    end

    if player.Team then
        local idx = teamMap[player.Team.Name]
        if idx then return idx end
    end

    if player.TeamColor then
        local c = player.TeamColor.Color
        for name, idx in pairs(teamMap) do
            if name == player.TeamColor.Name then
                return idx
            end
        end
    end
    return nil
end

local function getTeamName(player)
    if not player then return "?" end
    local attr = player:GetAttribute("Team")
    if attr and attr ~= "" then return tostring(attr) end
    if player.Team then return player.Team.Name end
    if player.TeamColor then return player.TeamColor.Name end
    return "?"
end

local function PaintToggle(btn, on, label, colorOn)
    if on then
        btn.Text = label .. ": ON"
        btn.BackgroundColor3 = colorOn or Colors.on
        btn.TextColor3 = Color3.new(1, 1, 1)
    else
        btn.Text = label .. ": OFF"
        btn.BackgroundColor3 = Colors.off
        btn.TextColor3 = Color3.new(0.75, 0.75, 0.75)
    end
end

local function MakeToggle(parent, label, order, getter, setter, colorOn)
    local btn = MakeButton(parent, label .. ": OFF", order, function()
        local newState = not getter()
        setter(newState)
        PaintToggle(btn, newState, label, colorOn)
        saveSettings()
    end)
    PaintToggle(btn, getter(), label, colorOn)
    return btn
end

local function stopStick()
    if stickState.connection then
        pcall(function() stickState.connection:Disconnect() end)
        stickState.connection = nil
    end
    stickState.active = false
    local char = LP.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then pcall(function() hum.PlatformStand = false end) end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                pcall(function() part.CanCollide = true end)
            end
        end
    end
end
